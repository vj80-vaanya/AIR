cmake_minimum_required(VERSION 3.20)
project(security_engine VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)

# Compiler flags
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
  add_compile_options(-fno-strict-aliasing)
endif()

# Build type defaults
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Find packages
find_package(OpenSSL 3.0 REQUIRED)
find_package(ZLIB REQUIRED)

# SQLite (bundled)
add_library(sqlite3 STATIC
  third_party/sqlite3/sqlite3.c
)
target_compile_definitions(sqlite3 PRIVATE
  SQLITE_HAS_CODEC
  SQLITE_THREADSAFE=1
  SQLITE_DEFAULT_WAL_SYNCHRONOUS=1
)

# ONNX Runtime C API — optional; set -DORT_ROOT=/path/to/onnxruntime to enable
# Android: use onnxruntime-android AAR (extracted headers + libonnxruntime.so)
# Desktop: download from https://github.com/microsoft/onnxruntime/releases
option(ENABLE_ORT "Enable ONNX Runtime ML inference" OFF)
if(ENABLE_ORT)
  find_library(ORT_LIB onnxruntime
    PATHS ${ORT_ROOT}/lib
    REQUIRED
  )
  if(NOT EXISTS "${ORT_ROOT}/include/onnxruntime_c_api.h")
    message(FATAL_ERROR "onnxruntime_c_api.h not found in ${ORT_ROOT}/include")
  endif()
  message(STATUS "ORT enabled: ${ORT_LIB}")
else()
  message(STATUS "ORT disabled — pattern-matching only (pass -DENABLE_ORT=ON -DORT_ROOT=/path to enable)")
endif()

# Core security engine shared library
set(ENGINE_SOURCES
  src/security_engine.c
  src/fall_detector.c
  src/sos_system.c
  src/encryption.c
  src/ml_inference.c
  src/phone_analyzer.c
  src/sms_analyzer.c
  src/email_analyzer.c
  src/whatsapp_analyzer.c
  src/pattern_matcher.c
  src/db_manager.c
  src/vocab_table.c    # generated by scripts/build_c_vocab.py
)

add_library(security_engine SHARED ${ENGINE_SOURCES})

target_include_directories(security_engine PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/include
  ${OPENSSL_INCLUDE_DIR}
  third_party/sqlite3
)

set(_ORT_LIBS "")
if(ENABLE_ORT)
  target_include_directories(security_engine PUBLIC ${ORT_ROOT}/include)
  target_compile_definitions(security_engine PRIVATE HAVE_ORT=1)
  set(_ORT_LIBS ${ORT_LIB})
endif()

target_link_libraries(security_engine PRIVATE
  sqlite3
  OpenSSL::SSL
  OpenSSL::Crypto
  ZLIB::ZLIB
  ${_ORT_LIBS}
  m
)

target_compile_definitions(security_engine PRIVATE
  SE_VERSION_MAJOR=1
  SE_VERSION_MINOR=0
  SE_VERSION_PATCH=0
)

# Tests
option(BUILD_TESTS "Build unit tests" ON)
if(BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

# Install
install(TARGETS security_engine
  LIBRARY DESTINATION lib
  ARCHIVE DESTINATION lib
)
install(FILES include/security_engine.h DESTINATION include)
